enumerate() и zip()

Две незаменимые функции для работы с циклами.

Python — enumerate()
fruits = ["яблоко", "банан", "вишня"]

# Без enumerate — ручной счётчик
for i in range(len(fruits)):
    print(i, fruits[i])

# С enumerate — элегантно!
for i, fruit in enumerate(fruits, start=1):
    print(f"{i}. {fruit}")
# 1. яблоко
# 2. банан
# 3. вишня
Python — zip()
names = ["Алихан", "Асель", "Тимур"]
scores = [85, 92, 78]

# zip объединяет два списка попарно
for name, score in zip(names, scores):
    print(f"{name}: {score}")

# Создание словаря из двух списков
result = dict(zip(names, scores))
print(result)  # {'Алихан': 85, 'Асель': 92, 'Тимур': 78}

Прочие полезные функции

Python
# isinstance() — проверка типа
print(isinstance(3.14, float))  # True
print(isinstance("abc", (str, int)))  # True

# any() / all()
nums = [2, 4, 6, 7, 8]
print(any(n % 2 != 0 for n in nums))  # True — есть нечётные
print(all(n > 0 for n in nums))       # True — все положительные

# map() — применить функцию к каждому элементу
doubled = list(map(lambda x: x * 2, [1, 2, 3]))
print(doubled)  # [2, 4, 6]

# filter() — фильтрация по условию
evens = list(filter(lambda x: x % 2 == 0, range(10)))
print(evens)  # [0, 2, 4, 6, 8]
