Итоговый проект 2: «Угадай число»

Объединяем всё, что изучили в курсе: случайные числа, циклы, исключения, файлы и классы — в одну полноценную игру.

Цель проекта: создать игру «Угадай число» с уровнями сложности, счётчиком очков, сохранением рекорда в файл и ООП-архитектурой.

Что будем строить

Игра «Угадай число» с расширенными возможностями:

  • Три уровня сложности (лёгкий 1–50, средний 1–100, сложный 1–500)
  • Подсказки «горячо/холодно» (насколько близко)
  • Подсчёт очков (меньше попыток = больше очков)
  • Сохранение рекорда в файл
  • Возможность сыграть ещё раз

Шаг 1: Структура — класс Game

Python — game.py
import random

LEVELS = {
    "1": ("Лёгкий",  1,  50,  10),  # название, мин, макс, лимит попыток
    "2": ("Средний", 1, 100,   7),
    "3": ("Сложный", 1, 500,   5),
}

class Game:
    def __init__(self, level_key):
        name, mn, mx, limit = LEVELS[level_key]
        self.level = name
        self.secret = random.randint(mn, mx)
        self.min_val = mn
        self.max_val = mx
        self.max_attempts = limit
        self.attempts = 0
        self.history = []

    def hint(self, guess):
        diff = abs(guess - self.secret)
        if diff == 0:
            return "ПОБЕДА"
        elif diff <= 5:
            return "Очень горячо!"
        elif diff <= 15:
            return "Горячо"
        elif diff <= 30:
            return "Тепло"
        else:
            return "Холодно"

    def score(self):
        return max(0, (self.max_attempts - self.attempts + 1) * 10)

    def play_turn(self, guess):
        self.attempts += 1
        self.history.append(guess)
        if guess < self.secret:
            return "Больше!", self.hint(guess)
        elif guess > self.secret:
            return "Меньше!", self.hint(guess)
        else:
            return "Угадал!", "ПОБЕДА"

Шаг 2: Сохранение рекорда

Python
RECORD_FILE = "record.txt"

def load_record():
    try:
        with open(RECORD_FILE, "r") as f:
            return int(f.read().strip())
    except (FileNotFoundError, ValueError):
        return 0

def save_record(score):
    with open(RECORD_FILE, "w") as f:
        f.write(str(score))
    print(f"Новый рекорд сохранён: {score}!")

Шаг 3: Главный игровой цикл

Python
def choose_level():
    print("\nУровни сложности:")
    for key, (name, mn, mx, lim) in LEVELS.items():
        print(f"  {key}. {name} ({mn}–{mx}, {lim} попыток)")
    while True:
        choice = input("Выбери уровень (1/2/3): ").strip()
        if choice in LEVELS:
            return choice
        print("Введи 1, 2 или 3")

def run_game():
    record = load_record()
    print(f"Текущий рекорд: {record} очков")

    while True:
        level_key = choose_level()
        game = Game(level_key)
        print(f"\nЯ загадал число от {game.min_val} до {game.max_val}.")
        print(f"У тебя {game.max_attempts} попыток. Удачи!\n")

        won = False
        while game.attempts < game.max_attempts:
            remaining = game.max_attempts - game.attempts
            try:
                guess = int(input(f"Попытка {game.attempts + 1}/{game.max_attempts}: "))
            except ValueError:
                print("Введи целое число!")
                continue

            direction, heat = game.play_turn(guess)
            if heat == "ПОБЕДА":
                won = True
                break
            print(f"  → {direction}  [{heat}]  (осталось {remaining - 1} попыток)")

        if won:
            s = game.score()
            print(f"\nПравильно! Число: {game.secret}")
            print(f"Попыток: {game.attempts}. Очки: {s}")
            if s > record:
                record = s
                save_record(record)
        else:
            print(f"\nПопытки закончились! Загаданное число: {game.secret}")

        again = input("\nСыграть ещё? (да/нет): ").strip().lower()
        if again not in ("да", "д", "yes", "y"):
            print("Спасибо за игру!")
            break

run_game()
